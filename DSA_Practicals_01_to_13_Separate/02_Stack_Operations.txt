// CODE 1

#include <iostream>
using namespace std;
int main() {
    int stack[5];
    int top = -1;
    if (top == 4) {
        cout << "Stack Overflow";
        return 0;
    }
    top++;
    stack[top] = 10;
    cout << "Pushed: " << stack[top];
    return 0;
}

// CODE 5

#include <iostream>
using namespace std;
const int SIZE = 5;
class Stack {
    int stack[SIZE];
    int top;
public:
    Stack() {
        top = -1;
    }
    void push(int value) {
        if (top == SIZE - 1) {
            cout << "Stack Overflow!\n";
            return;
        }
        stack[++top] = value;
        cout << value << " pushed.\n";
    }
    void pop() {
        if (top == -1) {
            cout << "Stack Underflow!\n";
            return;
        }
        cout << stack[top--] << " popped.\n";
    }
    void peek() {
        if (top == -1) {
            cout << "Stack is empty.\n";
            return;
        }
        cout << "Top element: " << stack[top] << "\n";
    }
    void display() {
        if (top == -1) {
            cout << "Stack is empty.\n";
            return;
        }
        cout << "Stack: ";
        for (int i = top; i >= 0; i--) {
            cout << stack[i] << " ";
        }
        cout << "\n";
    }
    void search(int value) {
        for (int i = top; i >= 0; i--) {
            if (stack[i] == value) {
                cout << value << " found at position " << i + 1 << ".\n";
                return;
            }
        }
        cout << value << " not found.\n";
    }
};
int main() {
    Stack s;
    s.push(10);
    s.push(20);
    s.push(30);
    s.display();
    s.peek();
    s.search(20);
    s.pop();
    s.display();
    return 0;
}
