// PRACTICAL 8: SINGLY LINKED LIST - CREATION, INSERTION AND DELETION

#include <iostream>
using namespace std;
struct Node {
    int data;
    Node* next;
};
Node* head = nullptr;
void create(int value) {
    Node* newNode = new Node{value, nullptr};
    if (head == nullptr) {
        head = newNode;
    } else {
        Node* temp = head;
        while (temp->next != nullptr) {
            temp = temp->next;
        }
        temp->next = newNode;
    }
}
void insert(int value) {
    Node* newNode = new Node{value, head};
    head = newNode;
}
void deleteNode() {
    if (head == nullptr) {
        cout << "List is empty.\n";
        return;
    }
    Node* temp = head;
    head = head->next;
    delete temp;
    cout << "First node deleted.\n";
}
void display() {
    if (head == nullptr) {
        cout << "List is empty.\n";
        return;
    }
    Node* temp = head;
    while (temp != nullptr) {
        cout << temp->data << " -> ";
        temp = temp->next;
    }
    cout << "NULL\n";
}
int main() {
    int choice, value;
    do {
        cout << "\n1. Create";
        cout << "\n2. Insert at Beginning";
        cout << "\n3. Delete from Beginning";
        cout << "\n4. Display";
        cout << "\n5. Exit";
        cout << "\nEnter choice: ";
        cin >> choice;
        switch (choice) {
            case 1:
                cout << "Enter value: ";
                cin >> value;
                create(value);
                break;
            case 2:
                cout << "Enter value: ";
                cin >> value;
                insert(value);
                break;
            case 3:
                deleteNode();
                break;
            case 4:
                display();
                break;
            case 5:
                cout << "Exit\n";
                break;
            default:
                cout << "Invalid choice\n";
        }
    } while (choice != 5);
    return 0;
}
