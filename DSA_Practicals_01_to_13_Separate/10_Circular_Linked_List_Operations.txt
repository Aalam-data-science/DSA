// PRACTICAL 10: CIRCULAR LINKED LIST

#include <iostream>
using namespace std;
class Node {
public:
    int data;
    Node* next;
    Node(int value) {
        data = value;
        next = nullptr;
    }
};
class CircularLinkedList {
    Node* head;
public:
    CircularLinkedList() {
        head = nullptr;
    }
    void insertBegin(int value) {
        Node* newNode = new Node(value);
        if (head == nullptr) {
            head = newNode;
            newNode->next = head;
            return;
        }
        Node* last = head;
        while (last->next != head) {
            last = last->next;
        }
        newNode->next = head;
        last->next = newNode;
        head = newNode;
    }
    void insertEnd(int value) {
        Node* newNode = new Node(value);
        if (head == nullptr) {
            head = newNode;
            newNode->next = head;
            return;
        }
        Node* last = head;
        while (last->next != head) {
            last = last->next;
        }
        last->next = newNode;
        newNode->next = head;
    }
    void display() const {
        if (head == nullptr) {
            cout << "List is empty.\n";
            return;
        }
        Node* temp = head;
        do {
            cout << temp->data << " -> ";
            temp = temp->next;
        } while (temp != head);
        cout << "(head)\n";
    }
    int search(int value) const {
        if (head == nullptr) {
            return -1;
        }
        Node* temp = head;
        int index = 0;
        do {
            if (temp->data == value) {
                return index;
            }
            temp = temp->next;
            index++;
        } while (temp != head);
        return -1;
    }
    void deleteValue(int value) {
        if (head == nullptr) {
            cout << "List is empty.\n";
            return;
        }
        if (head->data == value && head->next == head) {
            delete head;
            head = nullptr;
            cout << "Deleted " << value << ".\n";
            return;
        }
        if (head->data == value) {
            Node* last = head;
            while (last->next != head) {
                last = last->next;
            }
            Node* temp = head;
            head = head->next;
            last->next = head;
            delete temp;
            cout << "Deleted " << value << ".\n";
            return;
        }
        Node* previous = head;
        Node* current = head->next;
        while (current != head && current->data != value) {
            previous = current;
            current = current->next;
        }
        if (current == head) {
            cout << value << " not found.\n";
            return;
        }
        previous->next = current->next;
        delete current;
        cout << "Deleted " << value << ".\n";
    }
    void reverseList() {
        if (head == nullptr || head->next == head) {
            return;
        }
        Node* previous = nullptr;
        Node* current = head;
        Node* nextNode;
        do {
            nextNode = current->next;
            current->next = previous;
            previous = current;
            current = nextNode;
        } while (current != head);
        Node* oldHead = head;
        head = previous;
        oldHead->next = head;
    }
    void displayBackward() const {
        if (head == nullptr) {
            cout << "List is empty.\n";
            return;
        }
        Node* last = head;
        while (last->next != head) {
            last = last->next;
        }
        Node* current = last;
        cout << "Backward: ";
        do {
            cout << current->data << " ";
            Node* previous = head;
            while (previous->next != current) {
                previous = previous->next;
            }
            current = previous;
        } while (current != last);
        cout << endl;
    }
};
int main() {
    CircularLinkedList list;
    list.insertBegin(20);
    list.insertBegin(10);
    list.insertEnd(30);
    list.insertEnd(40);
    cout << "Initial Circular List: ";
    list.display();
    int target = 30;
    int index = list.search(target);
    if (index != -1) {
        cout << target << " found at 0-based index: " << index << endl;
    } else {
        cout << target << " not found." << endl;
    }
    cout << "Backward traversal: ";
    list.displayBackward();
    cout << "After deleting 20: ";
    list.deleteValue(20);
    list.display();
    list.reverseList();
    cout << "After reverse: ";
    list.display();
    return 0;
}
