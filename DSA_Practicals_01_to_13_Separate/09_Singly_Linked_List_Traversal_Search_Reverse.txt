// PRACTICAL 9: SINGLY LINKED LIST - TRAVERSAL, SEARCH AND REVERSE

#include <iostream>
using namespace std;
struct Node {
    int data;
    Node* next;
};
Node* head = nullptr;
void create(int n) {
    Node* tail = nullptr;
    for (int i = 0; i < n; i++) {
        Node* newNode = new Node;
        cout << "Enter element " << i + 1 << ": ";
        cin >> newNode->data;
        newNode->next = nullptr;
        if (head == nullptr) {
            head = tail = newNode;
        } else {
            tail->next = newNode;
            tail = newNode;
        }
    }
}
void traverse() {
    for (Node* p = head; p != nullptr; p = p->next) {
        cout << p->data << " ";
    }
    cout << endl;
}
void search(int value) {
    int position = 1;
    for (Node* p = head; p != nullptr; p = p->next, position++) {
        if (p->data == value) {
            cout << "Found at position " << position << endl;
            return;
        }
    }
    cout << "Not Found" << endl;
}
void reverseList() {
    Node* previous = nullptr;
    Node* current = head;
    while (current != nullptr) {
        Node* nextNode = current->next;
        current->next = previous;
        previous = current;
        current = nextNode;
    }
    head = previous;
}
int main() {
    int n, value;
    cout << "Enter number of nodes: ";
    cin >> n;
    if (n <= 0) {
        cout << "Number of nodes must be positive.";
        return 0;
    }
    create(n);
    cout << "List: ";
    traverse();
    cout << "Enter value to search: ";
    cin >> value;
    search(value);
    reverseList();
    cout << "Reversed List: ";
    traverse();
    return 0;
}
